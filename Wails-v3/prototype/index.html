+func (b *Bridge) Act(calls []ToolCall) []ToolResult {
+ var wg sync.WaitGroup
+ results := make([]ToolResult, len(calls))
+ for i, c := range calls {
+ wg.Add(1)
+ go func(idx int, call ToolCall) {
+ defer wg.Done()
+ // ReqId 幂等: 命中缓存直接返回
+ if cached, ok := b.idem.Get(call.ReqId); ok {
+ results[idx] = cached
+ return
+ }
+ resp, _ := b.callTool(ctx, call)
+ b.idem.Set(call.ReqId, resp)
+ results[idx] = resp
+ }(i, c)
+ }
+ wg.Wait()
+ return results
}

关键改动:用 sync.WaitGroup 并行派发,结果按索引保序;Go 侧增加 idem LRU 缓存基于 ReqId 去重,符合原架构"Erlang 生成全局 ReqId"的幂等设计。Erlang 侧无需改动,只需接收保序的结果切片。
